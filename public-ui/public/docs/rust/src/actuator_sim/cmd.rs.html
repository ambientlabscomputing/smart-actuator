1//! CLI client for the main `ActuatorService` wire endpoint.
2
3use anyhow::Context;
4use clap::Subcommand;
5
6use actuator_proto::actuator::{CommandResponse, ControlModeProto};
7use actuator_proto::wire::async_wire::WireClient;
8
9#[derive(Debug, Subcommand)]
10pub enum CmdCommand {
11    /// Read position, velocity, current and temperature in one shot
12    Status,
13    /// Set target position (rad)
14    Move {
15        /// Target angle in radians
16        angle: f64,
17    },
18    /// Set target velocity (rad/s)
19    Velocity { velocity: f64 },
20    /// Set target torque (Nm)
21    Torque { torque: f64 },
22    /// Set control mode [position | velocity | torque | impedance]
23    SetMode { mode: String },
24    /// Set soft position limits (rad)
25    SetLimits { min: f64, max: f64 },
26    /// Set maximum current limit (A)
27    SetCurrentLimit { max_current: f64 },
28    /// Set maximum temperature limit (deg C)
29    SetTempLimit { max_temperature: f64 },
30    /// Clear a latched fault
31    ClearFault,
32    /// Read the simulator clock (sim monotonic seconds)
33    Clock,
34    /// Report trajectory tracking error statistics
35    TrackingError,
36    /// Pause trajectory execution
37    Pause,
38    /// Resume a paused trajectory
39    Resume,
40    /// Abort and discard the active trajectory
41    Abort,
42}
43
44fn parse_mode(s: &str) -> anyhow::Result<ControlModeProto> {
45    match s.to_lowercase().as_str() {
46        "position" | "pos" => Ok(ControlModeProto::ControlModePosition),
47        "velocity" | "vel" => Ok(ControlModeProto::ControlModeVelocity),
48        "torque" => Ok(ControlModeProto::ControlModeTorque),
49        "impedance" => Ok(ControlModeProto::ControlModeImpedance),
50        _ => anyhow::bail!("Unknown mode '{s}'. Valid: position, velocity, torque, impedance"),
51    }
52}
53
54fn print_cmd(r: CommandResponse) {
55    if r.success {
56        println!("ok");
57    } else {
58        println!("refused [{}]: {}", refusal_name(r.refusal_code), r.message);
59    }
60}
61
62fn refusal_name(code: i32) -> &'static str {
63    match code {
64        0 => "none",
65        1 => "OutsideSoftLimits",
66        2 => "WrongControlMode",
67        3 => "OverTemperature",
68        4 => "FaultLatched",
69        5 => "NotImplemented",
70        6 => "TrajectoryRunning",
71        7 => "InvalidTrajectory",
72        _ => "unknown",
73    }
74}
75
76pub async fn execute(addr: String, cmd: CmdCommand) -> anyhow::Result<()> {
77    let c = WireClient::connect(&addr)
78        .await
79        .context("Cannot connect to actuator service")?;
80
81    match cmd {
82        CmdCommand::Status => {
83            let pos = c.read_position().await.context("read_position")?;
84            let vel = c.read_velocity().await.context("read_velocity")?;
85            let cur = c.read_current().await.context("read_current")?;
86            let tmp = c.read_temperature().await.context("read_temperature")?;
87            println!("position:    {:>12.4} rad", pos.angle);
88            println!("velocity:    {:>12.4} rad/s", vel.velocity);
89            println!("current:     {:>12.4} A", cur.current);
90            println!("temperature: {:>12.4} C", tmp.temperature);
91        }
92        CmdCommand::Move { angle } => {
93            print_cmd(c.set_position(angle).await.context("set_position")?);
94        }
95        CmdCommand::Velocity { velocity } => {
96            print_cmd(c.set_velocity(velocity).await.context("set_velocity")?);
97        }
98        CmdCommand::Torque { torque } => {
99            print_cmd(c.set_torque(torque).await.context("set_torque")?);
100        }
101        CmdCommand::SetMode { mode } => {
102            let m = parse_mode(&mode)? as i32;
103            print_cmd(c.set_control_mode(m).await.context("set_control_mode")?);
104        }
105        CmdCommand::SetLimits { min, max } => {
106            print_cmd(c.set_soft_limits(min, max).await.context("set_soft_limits")?);
107        }
108        CmdCommand::SetCurrentLimit { max_current } => {
109            print_cmd(c.set_current_limit(max_current).await.context("set_current_limit")?);
110        }
111        CmdCommand::SetTempLimit { max_temperature } => {
112            print_cmd(
113                c.set_temperature_limit(max_temperature)
114                    .await
115                    .context("set_temperature_limit")?,
116            );
117        }
118        CmdCommand::ClearFault => {
119            print_cmd(c.clear_fault().await.context("clear_fault")?);
120        }
121        CmdCommand::Clock => {
122            let t = c.get_clock().await.context("get_clock")?;
123            println!("sim_time: {:>14.6} s", t);
124        }
125        CmdCommand::TrackingError => {
126            let r = c.report_tracking_error().await.context("report_tracking_error")?;
127            println!("instantaneous:   {:>12.6} rad", r.instantaneous);
128            println!("max_since_start: {:>12.6} rad", r.max_since_start);
129            println!("rms:             {:>12.6} rad", r.rms);
130        }
131        CmdCommand::Pause => {
132            print_cmd(c.pause().await.context("pause")?);
133        }
134        CmdCommand::Resume => {
135            print_cmd(c.resume().await.context("resume")?);
136        }
137        CmdCommand::Abort => {
138            print_cmd(c.abort().await.context("abort")?);
139        }
140    }
141
142    Ok(())
143}
