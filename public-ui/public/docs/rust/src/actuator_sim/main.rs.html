17#[derive(Parser)]
18#[command(name = "actuator-sim", about = "Ambient Labs actuator simulator")]
19struct Cli {
20    #[command(subcommand)]
21    command: Commands,
22}
23
24#[derive(Subcommand)]
25enum Commands {
26    /// Run the actuator simulator gRPC service
27    Run {
28        /// Detach from the terminal and run in the background.
29        /// Requires log_to_stderr = false in config so logs go to the log file.
30        #[arg(short, long)]
31        detached: bool,
32    },
33    /// Stop a running actuator simulator by sending SIGTERM via its PID file
34    Stop,
35    /// Send a command to the actuator service (port from config)
36    Cmd {
37        #[command(subcommand)]
38        command: cmd::CmdCommand,
39    },
40    /// Interact with the simulator backdoor (port from config)
41    Sim {
42        #[command(subcommand)]
43        command: sim_cmd::SimCommand,
44    },
45}

47/// Outer main is intentionally synchronous: the fork for detached mode must
48/// happen before the Tokio runtime is created (forking a multi-threaded process
49/// is unsafe).
50fn main() -> anyhow::Result<()> {
51    let cli = Cli::parse();
52
53    // Config is always loaded first — both Run and Stop need it.
54    let config_path = config::config_path();
55    let config = config::load(&config_path)?;
56
57    match cli.command {
58        Commands::Run { detached } => {
59            if detached {
60                detach(&config, &config_path)?;
61            }
62            tokio::runtime::Builder::new_multi_thread()
63                .enable_all()
64                .build()?
65                .block_on(run(config, config_path))
66        }
67        Commands::Stop => stop(&config.pid_file),
68        Commands::Cmd { command } => {
69            // Wire protocol uses plain TCP — strip the http:// scheme.
70            let addr = format!("{}:{}", config.grpc.host, config.grpc.port);
71            tokio::runtime::Builder::new_current_thread()
72                .enable_all()
73                .build()?
74                .block_on(cmd::execute(addr, command))
75        }
76        Commands::Sim { command } => {
77            let addr = format!("http://{}:{}", config.grpc.host, config.grpc.backdoor_port);
78            tokio::runtime::Builder::new_current_thread()
79                .enable_all()
80                .build()?
81                .block_on(sim_cmd::execute(addr, command))
82        }
83    }
84}

86/// Fork and detach from the controlling terminal (Unix only).
87#[cfg(unix)]
88fn detach(config: &config::ActuatorConfig, config_path: &str) -> anyhow::Result<()> {
89    use daemonize::{Daemonize, Outcome};
90    let cwd = std::env::current_dir()?;
91
92    let log_sink = if config.log_settings.log_to_stderr {
93        "stderr (warning: will be lost after detach)".to_owned()
94    } else {
95        config.log_settings.file.clone()
96    };
97    let backdoor_line = if config.grpc.enable_backdoor {
98        format!("{}:{}", config.grpc.host, config.grpc.backdoor_port)
99    } else {
100        "disabled".to_owned()
101    };
102    let msg = format!(
103        "actuator-sim detached\n  identity : {} ({})\n  gRPC     : {}:{}\n  backdoor : {}\n  log      : {}\n  pid file : {}\n  config   : {}",
104        config.identity.name,
105        config.identity.id,
106        config.grpc.host,
107        config.grpc.port,
108        backdoor_line,
109        log_sink,
110        config.pid_file,
111        config_path,
112    );
113
114    match Daemonize::new().working_directory(cwd).execute() {
115        Outcome::Parent(Ok(_)) => {
116            eprintln!("{msg}");
117            std::process::exit(0);
118        }
119        Outcome::Parent(Err(e)) => anyhow::bail!("Failed to detach: {e}"),
120        Outcome::Child(Ok(_)) => Ok(()), // child continues into run()
121        Outcome::Child(Err(e)) => anyhow::bail!("Failed to initialize daemon: {e}"),
122    }
123}

130/// Removes the PID file when dropped (i.e. on any exit path).
131struct PidGuard(String);
132impl Drop for PidGuard {
133    fn drop(&mut self) {
134        let _ = std::fs::remove_file(&self.0);
135    }
136}

138/// Send SIGTERM to the process named in the PID file (Unix only).
139#[cfg(unix)]
140fn stop(pid_file: &str) -> anyhow::Result<()> {
141    use nix::sys::signal::{Signal, kill};
142    use nix::unistd::Pid;
143
144    let raw = std::fs::read_to_string(pid_file).map_err(|_| {
145        anyhow::anyhow!("PID file not found at '{pid_file}'\nIs actuator-sim running?")
146    })?;
147    let pid: i32 = raw.trim().parse().map_err(|_| {
148        anyhow::anyhow!("Malformed PID file '{pid_file}': {:?}", raw.trim())
149    })?;
150
151    kill(Pid::from_raw(pid), Signal::SIGTERM)
152        .map_err(|e| anyhow::anyhow!("Failed to send SIGTERM to process {pid}: {e}"))?;
153
154    eprintln!("actuator-sim: sent SIGTERM to process {pid}");
155    Ok(())
156}

160    anyhow::bail!("stop is not supported on this platform")
161}

163async fn run(config: config::ActuatorConfig, config_path: String) -> anyhow::Result<()> {
164    // Keep the guard alive for the process lifetime — dropping it flushes the log sink.
165    let _log_guard = logging::setup(&config.log_settings);
166
167    info!(
168        config = config_path,
169        id = config.identity.id,
170        name = config.identity.name,
171        "actuator simulator starting"
172    );
173
174    // Build the SimPlant from config
175    let params = PlantParams {
176        inertia: config.physics.inertia,
177        damping: config.physics.damping,
178        encoder_noise_std: config.physics.encoder_noise_std,
179        kp_pos: config.physics.kp_pos,
180        kd_pos: config.physics.kd_pos,
181        kp_vel: config.physics.kp_vel,
182        kt: config.physics.kt,
183        thermal_resistance: config.physics.thermal_resistance,
184        thermal_capacitance: config.physics.thermal_capacitance,
185    };
186    let plant = SimPlant::new(params);
187
188    // AppService wraps the plant via the Hardware trait
189    let service = Arc::new(AppService::new(plant.clone()));
190    service.start().await;
191
192    // Write PID file; guard removes it on any exit path (clean shutdown or panic).
193    std::fs::write(&config.pid_file, format!("{}", std::process::id()))
194        .map_err(|e| anyhow::anyhow!("Cannot write PID file '{}': {e}", config.pid_file))?;
195    let _pid_guard = PidGuard(config.pid_file.clone());
196    info!(pid_file = config.pid_file, "PID file written");
197
198    // Physics tick task — runs AppService::tick then SimPlant::tick at `tick_hz`
199    let tick_dt = 1.0 / config.physics.tick_hz;
200    {
201        let tick_service = service.clone();
202        let tick_plant = plant.clone();
203        tokio::spawn(async move {
204            let mut interval = tokio::time::interval(Duration::from_secs_f64(tick_dt));
205            interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
206            loop {
207                interval.tick().await;
208                // 1. Advance control state and apply setpoint to plant
209                tick_service.tick(tick_dt).await;
210                // 2. Integrate plant dynamics
211                tick_plant.tick(tick_dt).await;
212            }
213        });
214    }
215
216    // Optional backdoor server
217    if config.grpc.enable_backdoor {
218        let bd_plant = plant.clone();
219        let bd_addr = format!("{}:{}", config.grpc.host, config.grpc.backdoor_port);
220        tokio::spawn(async move {
221            if let Err(e) = backdoor::serve(bd_plant, bd_addr, std::future::pending()).await {
222                tracing::error!("Backdoor server error: {e}");
223            }
224        });
225    }
226
227    // Main gRPC server — shuts down on SIGINT or SIGTERM (SIGTERM is sent by `stop`).
228    let service_ref: Arc<dyn actuator_core::Service> = service.clone();
229    let shutdown = async {
230        #[cfg(unix)]
231        {
232            use tokio::signal::unix::{SignalKind, signal as unix_signal};
233            let mut sigterm = unix_signal(SignalKind::terminate())
234                .expect("failed to install SIGTERM handler");
235            let mut sigint = unix_signal(SignalKind::interrupt())
236                .expect("failed to install SIGINT handler");
237            tokio::select! {
238                _ = sigterm.recv() => info!("SIGTERM received"),
239                _ = sigint.recv()  => info!("SIGINT received"),
240            }
241        }
242        #[cfg(not(unix))]
243        {
244            signal::ctrl_c().await.expect("failed to install Ctrl-C handler");
245            info!("shutdown signal received");
246        }
247    };
248
249    server::serve(&config.grpc, service_ref, shutdown).await?;
250
251    service.stop().await;
252    info!("shutdown complete");
253
254    Ok(())
255}
